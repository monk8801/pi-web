# 任务列表（待办）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 Pi Web 里提供一份人工维护的待办清单：按项目分组，一条待办点一下就进入该项目开跑一个 pi 会话，agent 收尾时显式宣告完成（否则退回人工确认）。

**Architecture:** 功能主体全部落在新文件里：`lib/todos-shared.ts`（纯类型与规则）→ `lib/todos.ts`（`pi-web-todos.json` 的加锁原子读写）→ `app/api/todos/route.ts`（GET/POST，含 `start`）→ `hooks/useTodos.ts` + `components/TodoPanel.tsx`（右栏面板）。对上游文件只做加法：一个内置扩展注册无参工具 `todo_complete`（`lib/builtin-todo-extension.ts` + `lib/builtin-extensions.ts` + `lib/rpc-manager.ts` 各一处），侧栏 footer 一行入口（`lib/session-tree.ts` / `components/SessionTree.tsx` / `components/SessionSidebar.tsx`），右栏 tab 一类（`components/TabBar.tsx` / `components/AppShell.tsx`），三份 i18n 文案。

**Tech Stack:** Next.js 16 App Router（route handlers）、TypeScript、React 19（客户端组件）、`proper-lockfile` + `lib/atomic-file.ts`（存储）、node:test + jiti + `react-dom/server`（测试）。

```ts
// lib/todos-shared.ts
// 人工待办的类型与纯规则。客户端与服务端共用，因此不引入任何 Node 模块。
// 服务端把状态存在 `pi-web-todos.json`（`lib/todos.ts`），本文件只放两边界都能跑的规则。

export const TODOS_VERSION = 1;
export const MAX_TODOS = 2000;
export const MAX_TODO_TITLE_LENGTH = 200;
export const MAX_TODO_NOTE_LENGTH = 4000;
export const MAX_PROJECT_LENGTH = 4096;
const TODO_ID_PATTERN = /^[0-9a-f]{8}$/;

/** 只持久化两态；「进行中」「待确认」由会话状态派生。 */
export type TodoStatus = "open" | "done";
export type TodoDoneBy = "agent" | "user";
/** 展示态：未开始 / 进行中 / 待确认 / 已完成。 */
export type TodoDisplayStatus = "open" | "running" | "confirm" | "done";

export interface Todo {
  id: string;
  title: string;
  note?: string;
  /** 绝对路径：待办所属项目，也是开跑会话的 cwd。 */
  project: string;
  /** 最近一次执行的 pi 会话 id。 */
  sessionId?: string;
  createdAt: number;
  updatedAt: number;
  doneAt?: number;
  doneBy?: TodoDoneBy;
  status: TodoStatus;
}

export interface TodosState {
  version: typeof TODOS_VERSION;
  /** 单调递增，服务端每次真正改变文件的写入 +1。 */
  revision: number;
  todos: Todo[];
}

export interface TodoAddRequest { action: "add"; title: string; note?: string; project: string }
export interface TodoUpdateRequest { action: "update"; id: string; title?: string; note?: string; project?: string }
export interface TodoIdRequest { action: "complete" | "reopen" | "delete"; id: string }
/** 服务端在 start 之后回填会话 id，客户端不发它。 */
export interface TodoAttachRequest { action: "attach"; id: string; sessionId: string }
export type TodoRequest =
  | TodoAddRequest
  | TodoUpdateRequest
  | TodoIdRequest
  | TodoAttachRequest;

export type TodoRequestParse =
  | { ok: true; request: TodoRequest }
  | { ok: false; error: string };

export function emptyTodosState(): TodosState {
  return { version: TODOS_VERSION, revision: 0, todos: [] };
}

function trimmed(value: unknown): string | undefined {
  if (typeof value !== "string") return undefined;
  const text = value.trim();
  return text.length > 0 ? text : undefined;
}

function validId(value: unknown): value is string {
  return typeof value === "string" && TODO_ID_PATTERN.test(value);
}

function validTodo(raw: unknown): Todo | null {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) return null;
  const entry = raw as Record<string, unknown>;
  const id = entry.id;
  const title = trimmed(entry.title);
  const project = trimmed(entry.project);
  const status = entry.status === "done" ? "done" : entry.status === "open" ? "open" : null;
  if (!validId(id) || title === undefined || project === undefined || status === null) return null;
  const createdAt = typeof entry.createdAt === "number" && Number.isFinite(entry.createdAt) ? entry.createdAt : 0;
  const updatedAt = typeof entry.updatedAt === "number" && Number.isFinite(entry.updatedAt) ? entry.updatedAt : createdAt;
  const note = trimmed(entry.note);
  const sessionId = trimmed(entry.sessionId);
  const doneAt = typeof entry.doneAt === "number" && Number.isFinite(entry.doneAt) ? entry.doneAt : undefined;
  const doneBy = entry.doneBy === "agent" ? "agent" : entry.doneBy === "user" ? "user" : undefined;
  return {
    id,
    title: title.slice(0, MAX_TODO_TITLE_LENGTH),
    ...(note ? { note: note.slice(0, MAX_TODO_NOTE_LENGTH) } : {}),
    project: project.slice(0, MAX_PROJECT_LENGTH),
    ...(sessionId ? { sessionId } : {}),
    createdAt,
    updatedAt,
    ...(status === "done" && doneAt !== undefined ? { doneAt } : {}),
    ...(status === "done" && doneBy ? { doneBy } : {}),
    status,
  };
}

/** 读文件时用：坏条目不信任，整个文档不可解析时返回 null（调用方按空状态处理）。 */
export function normalizeTodosState(raw: unknown): TodosState | null {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) return null;
  const document = raw as Record<string, unknown>;
  const revision = typeof document.revision === "number" && Number.isFinite(document.revision) && document.revision >= 0
    ? Math.floor(document.revision)
    : 0;
  const list = Array.isArray(document.todos) ? document.todos : [];
  const todos: Todo[] = [];
  for (const entry of list) {
    const todo = validTodo(entry);
    if (todo) todos.push(todo);
    if (todos.length >= MAX_TODOS) break;
  }
  return { version: TODOS_VERSION, revision, todos };
}

export function parseTodoRequest(raw: unknown): TodoRequestParse {
  if (typeof raw !== "object" || raw === null || Array.isArray(raw)) return { ok: false, error: "Expected a JSON object" };
  const body = raw as Record<string, unknown>;
  const action = body.action;
  if (action === "add") {
    const title = trimmed(body.title);
    const project = trimmed(body.project);
    if (!title) return { ok: false, error: "title is required" };
    if (!project) return { ok: false, error: "project is required" };
    const note = trimmed(body.note);
    return {
      ok: true,
      request: {
        action: "add",
        title: title.slice(0, MAX_TODO_TITLE_LENGTH),
        ...(note ? { note: note.slice(0, MAX_TODO_NOTE_LENGTH) } : {}),
        project: project.slice(0, MAX_PROJECT_LENGTH),
      },
    };
  }
  if (action === "update") {
    if (!validId(body.id)) return { ok: false, error: "id is required" };
    const title = trimmed(body.title);
    const note = trimmed(body.note);
    const project = trimmed(body.project);
    if (title === undefined && note === undefined && project === undefined) {
      return { ok: false, error: "nothing to update" };
    }
    return {
      ok: true,
      request: {
        action: "update",
        id: body.id,
        ...(title ? { title: title.slice(0, MAX_TODO_TITLE_LENGTH) } : {}),
        ...(note ? { note: note.slice(0, MAX_TODO_NOTE_LENGTH) } : {}),
        ...(project ? { project: project.slice(0, MAX_PROJECT_LENGTH) } : {}),
      },
    };
  }
  if (action === "complete" || action === "reopen" || action === "delete") {
    if (!validId(body.id)) return { ok: false, error: "id is required" };
    return { ok: true, request: { action, id: body.id } };
  }
  if (action === "attach") {
    if (!validId(body.id)) return { ok: false, error: "id is required" };
    const sessionId = trimmed(body.sessionId);
    if (!sessionId) return { ok: false, error: "sessionId is required" };
    return { ok: true, request: { action: "attach", id: body.id, sessionId } };
  }
  return { ok: false, error: "Unknown action" };
}

/** 一次变化：`changed` 为 false 时不写文件，revision 也不动。 */
export function applyTodoRequest(
  state: TodosState,
  request: TodoRequest,
  context: { now: number; newId?: () => string; doneBy?: TodoDoneBy },
): TodosState {
  const { now } = context;
  if (request.action === "add") {
    if (state.todos.length >= MAX_TODOS) return state;
    const id = (context.newId ?? randomTodoId)();
    const todo: Todo = {
      id,
      title: request.title,
      ...(request.note ? { note: request.note } : {}),
      project: request.project,
      createdAt: now,
      updatedAt: now,
      status: "open",
    };
    return { ...state, todos: [...state.todos, todo] };
  }
  if (request.action === "delete") {
    const todos = state.todos.filter((todo) => todo.id !== request.id);
    return todos.length === state.todos.length ? state : { ...state, todos };
  }
  const index = state.todos.findIndex((todo) => todo.id === request.id);
  if (index === -1) return state;
  const current = state.todos[index];
  let next: Todo;
  if (request.action === "update") {
    next = {
      ...current,
      ...(request.title ? { title: request.title } : {}),
      ...(request.note !== undefined ? { note: request.note } : {}),
      ...(request.project ? { project: request.project } : {}),
      updatedAt: now,
    };
  } else if (request.action === "attach") {
    if (current.sessionId === request.sessionId) return state;
    next = { ...current, sessionId: request.sessionId, updatedAt: now };
  } else if (request.action === "complete") {
    if (current.status === "done") return state;
    next = { ...current, status: "done", doneAt: now, doneBy: context.doneBy ?? "user", updatedAt: now };
  } else {
    if (current.status === "open") return state;
    const { doneAt: _doneAt, doneBy: _doneBy, ...rest } = current;
    next = { ...rest, status: "open", updatedAt: now };
  }
  const todos = [...state.todos];
  todos[index] = next;
  return { ...state, todos };
}

/** 会话在不在 running 集合里，决定「进行中」还是「待确认」。 */
export function todoDisplayStatus(todo: Todo, runningSessionIds: ReadonlySet<string>): TodoDisplayStatus {
  if (todo.status === "done") return "done";
  if (!todo.sessionId) return "open";
  return runningSessionIds.has(todo.sessionId) ? "running" : "confirm";
}

/** 开跑时发出的第一条消息：说清任务，并要求 agent 收尾时调用 todo_complete。 */
export function todoStartPrompt(todo: Pick<Todo, "id" | "title" | "note">): string {
  const lines = [`处理这个待办：`, "", todo.title];
  if (todo.note) lines.push("", todo.note);
  lines.push("", "完成之后调用 todo_complete 工具，把这条待办标记为已完成。");
  return lines.join("\n");
}

function randomTodoId(): string {
  const bytes = new Uint8Array(4);
  globalThis.crypto.getRandomValues(bytes);
  return [...bytes].map((byte) => byte.toString(16).padStart(2, "0")).join("");
}

// lib/todos.ts
// 待办清单存在 `pi-web-todos.json`（agent 目录下，与 mcp.json、settings.json 同级）。
// 它记录的是用户输入，所以不重建：写入在进程内排队、跨进程加锁、原子替换，
// 读不出来的文件被移到一边而不是被覆盖。规则在 `lib/todos-shared.ts`。

import { mkdirSync, renameSync, statSync } from "node:fs";
import { dirname, join } from "node:path";
import { getAgentDir } from "@earendil-works/pi-coding-agent";
import lockfile from "proper-lockfile";
import { writePrivateFileAtomicSync } from "./atomic-file";
import { serializeByKey } from "./key-serializer";
import { readRegularFileText } from "./regular-file";
import {
  applyTodoRequest,
  emptyTodosState,
  normalizeTodosState,
  TODOS_VERSION,
  type TodoDoneBy,
  type TodoRequest,
  type TodosState,
} from "./todos-shared";

const MAX_BYTES = 4 * 1024 * 1024;
const KNOWN_KEYS = new Set(["version", "revision", "todos", "__proto__"]);
const QUEUE_KEY: symbol = Symbol.for("pi-web:todos-write-queue");

const LOCK_OPTIONS = {
  realpath: false,
  stale: 10_000,
  retries: { retries: 10, factor: 2, minTimeout: 25, maxTimeout: 500 },
  onCompromised: (error: Error) => {
    console.warn(`[pi-web] the todos lock was compromised: ${error.message}`);
  },
};

export class TodosLockedError extends Error {
  constructor(message = "Another process holds the todos lock") {
    super(message);
    this.name = "TodosLockedError";
  }
}

interface TodosCache {
  /** 每个路径见过的最高 revision，文件被移走后 revision 仍向前走。 */
  revisions: Map<string, number>;
}

declare global {
  var __piWebTodosCache: TodosCache | undefined;
}

function cache(): TodosCache {
  return (globalThis.__piWebTodosCache ??= { revisions: new Map() });
}

function noteRevision(path: string, revision: number): void {
  const revisions = cache().revisions;
  if (revision > (revisions.get(path) ?? -1)) revisions.set(path, revision);
}

function errorCode(error: unknown): string | undefined {
  return (error as NodeJS.ErrnoException | undefined)?.code;
}

export function getTodosPath(agentDir = getAgentDir()): string {
  return join(agentDir, "pi-web-todos.json");
}

type Parsed = { ok: true; state: TodosState; raw: Record<string, unknown> } | { ok: false };

function parseStateText(text: string): Parsed {
  let raw: unknown;
  try {
    raw = JSON.parse(text);
  } catch {
    return { ok: false };
  }
  const state = normalizeTodosState(raw);
  return state ? { ok: true, state, raw: raw as Record<string, unknown> } : { ok: false };
}

/** 从未抛出的读：缺失或损坏都算空。 */
export function readTodos(path?: string): TodosState {
  let target: string;
  try {
    target = path ?? getTodosPath();
  } catch (error) {
    console.warn(`[pi-web] could not locate the todos file: ${error instanceof Error ? error.message : String(error)}`);
    return emptyTodosState();
  }
  const text = readRegularFileText(target, MAX_BYTES);
  if (text === undefined) return emptyTodosState();
  const parsed = parseStateText(text);
  if (parsed.ok) {
    noteRevision(target, parsed.state.revision);
    return parsed.state;
  }
  console.warn(`[pi-web] ${target} is not valid todos state; reading it as empty until the next change sets it aside`);
  return emptyTodosState();
}

function backUpCorruptFile(path: string): void {
  const backup = `${path}.corrupt-${Date.now()}`;
  try {
    renameSync(path, backup);
    console.warn(`[pi-web] ${path} could not be parsed; moved it to ${backup} and started from an empty todos list`);
  } catch (error) {
    console.warn(`[pi-web] ${path} could not be parsed and could not be moved aside (${error instanceof Error ? error.message : String(error)}); starting from an empty todos list`);
  }
}

export interface UpdateTodosOptions {
  path?: string;
  now?: number;
  newId?: () => string;
  /** `complete` 的署名：人点的为 "user"，agent 调工具为 "agent"。 */
  doneBy?: TodoDoneBy;
}

/** 加锁读改写；真正改变时 revision +1 并原子替换文件。 */
export function updateTodos(request: TodoRequest, options: UpdateTodosOptions = {}): Promise<TodosState> {
  let target: string;
  try {
    target = options.path ?? getTodosPath();
  } catch (error) {
    return Promise.reject(error);
  }
  const now = options.now ?? Date.now();
  return serializeByKey(QUEUE_KEY, target, async () => {
    mkdirSync(dirname(target), { recursive: true });
    let release: () => Promise<void>;
    try {
      release = await lockfile.lock(target, LOCK_OPTIONS);
    } catch (error) {
      if (errorCode(error) === "ELOCKED") throw new TodosLockedError();
      throw error;
    }
    try {
      const lastRevision = cache().revisions.get(target) ?? 0;
      const text = readRegularFileText(target, MAX_BYTES);
      let state: TodosState;
      const extras: Record<string, unknown> = {};
      let reset = false;
      if (text === undefined) {
        state = { ...emptyTodosState(), revision: lastRevision };
      } else {
        const parsed = parseStateText(text);
        if (parsed.ok) {
          state = parsed.state;
          for (const [key, value] of Object.entries(parsed.raw)) {
            if (!KNOWN_KEYS.has(key)) extras[key] = value;
          }
        } else {
          backUpCorruptFile(target);
          state = { ...emptyTodosState(), revision: lastRevision + 1 };
          reset = true;
        }
      }
      const next = applyTodoRequest(state, request, {
        now,
        ...(options.newId ? { newId: options.newId } : {}),
        ...(options.doneBy ? { doneBy: options.doneBy } : {}),
      });
      const changed = next !== state;
      if (!changed && !reset) return state;
      const document = {
        version: TODOS_VERSION,
        revision: state.revision + (changed ? 1 : 0),
        todos: next.todos,
        ...extras,
      };
      writePrivateFileAtomicSync(target, JSON.stringify(document, null, 2) + "\n");
      noteRevision(target, document.revision);
      return { version: TODOS_VERSION, revision: document.revision, todos: next.todos };
    } finally {
      try {
        await release();
      } catch {
        // A compromised lock was already reported through onCompromised.
      }
    }
  });
}

/** 供 `GET` 用的便宜读（文件的 mtime/size 不参与：一次读一次解析）。 */
export function getTodosRevision(path?: string): number | null {
  try {
    const target = path ?? getTodosPath();
    try {
      statSync(target);
    } catch (error) {
      const code = errorCode(error);
      if (code === "ENOENT" || code === "ENOTDIR") return 0;
      return null;
    }
    return readTodos(target).revision;
  } catch {
    return null;
  }
}

export function resetTodosCacheForTests(): void {
  globalThis.__piWebTodosCache = undefined;
}

// app/api/todos/route.ts
// 人工待办清单：GET 读全部，POST 一次一个动作（lib/todos.ts 存、lib/todos-shared.ts 校验）。
// 「开跑」在这一层完成：在本机新建一个 pi 会话并把任务内容作为首条消息发出（见 startTodo）。

import { NextResponse } from "next/server";
import { hasJsonContentType, isApiRequestAllowed } from "@/lib/request-security";
import { readTodos, TodosLockedError, updateTodos } from "@/lib/todos";
import { parseTodoRequest } from "@/lib/todos-shared";
import { isStartRequest, startTodo } from "@/lib/todos-start";

export const dynamic = "force-dynamic";

const NO_STORE = { "Cache-Control": "no-store" };

function refusal(status: number, reason: string, error: string) {
  return NextResponse.json({ error, reason }, { status, headers: NO_STORE });
}

// GET /api/todos
export async function GET() {
  const state = readTodos();
  return NextResponse.json({ todos: state.todos, revision: state.revision }, { headers: NO_STORE });
}

// POST /api/todos - one change; answers with the list after it.
export async function POST(req: Request) {
  if (!isApiRequestAllowed(req)) return refusal(403, "request-denied", "Untrusted API request");
  if (!hasJsonContentType(req)) return refusal(415, "content-type", "Content-Type must be application/json");

  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return refusal(400, "invalid-request", "Invalid JSON body");
  }
  const parsed = parseTodoRequest(body);
  if (!parsed.ok) return refusal(400, "invalid-request", parsed.error);

  const { request } = parsed;
  try {
    if (isStartRequest(request)) {
      const result = await startTodo(request.id);
      if (!result.ok) return refusal(result.status, result.reason, result.error);
      return NextResponse.json({ todos: result.state.todos, revision: result.state.revision, sessionId: result.sessionId }, { headers: NO_STORE });
    }
    if ("id" in request) {
      const known = readTodos().todos.some((todo) => todo.id === request.id);
      if (!known) return refusal(404, "not-found", "No such todo");
    }
    const state = await updateTodos(request, { doneBy: "user" });
    return NextResponse.json({ todos: state.todos, revision: state.revision }, { headers: NO_STORE });
  } catch (error) {
    if (error instanceof TodosLockedError) return refusal(409, "locked", error.message);
    return refusal(500, "write-failed", error instanceof Error ? error.message : String(error));
  }
}

// lib/builtin-todo-extension.ts
// 一个只属于「从待办开跑的会话」的内置扩展：注册无参工具 todo_complete。
// 它由该会话专属构造，todoId 闭包带入，因此不进入 toolNames 白名单，
// 也不会把用户选的工具预设钉死（设计文档 §6.2）。

import { defineTool, type InlineExtension } from "@earendil-works/pi-coding-agent";
import { Type } from "@earendil-works/pi-ai";
import { getTodosPath, readTodos, updateTodos } from "./todos";

export const TODO_COMPLETE_TOOL_NAME = "todo_complete";
export const TODO_COMPLETE_EXTENSION_NAME = "pi-web-todos";

/**
 * @param todoId 这条会话代表的待办。
 * @param path 仅测试用：覆盖待办文件的位置。
 */
export function createTodoCompleteExtension(options: { todoId: string; path?: string }): InlineExtension {
  return {
    name: TODO_COMPLETE_EXTENSION_NAME,
    hidden: true,
    factory: (pi) => {
      pi.registerTool(defineTool({
        name: TODO_COMPLETE_TOOL_NAME,
        label: "Complete todo",
        description: "Mark the todo this session was started from as done. Call it once, after the work is finished and verified.",
        parameters: Type.Object({}, { additionalProperties: false }),
        async execute() {
          const path = options.path ?? getTodosPath();
          const known = readTodos(path).todos.find((todo) => todo.id === options.todoId);
          if (!known) {
            return {
              content: [{ type: "text" as const, text: `Todo not found: ${options.todoId}` }],
              details: undefined,
              isError: true,
            };
          }
          if (known.status === "done") {
            return {
              content: [{ type: "text" as const, text: `Todo ${options.todoId} is already done` }],
              details: undefined,
            };
          }
          const state = await updateTodos({ action: "complete", id: options.todoId }, { path, doneBy: "agent" });
          const todo = state.todos.find((entry) => entry.id === options.todoId);
          return {
            content: [{ type: "text" as const, text: `Todo ${options.todoId} is done (${todo?.title ?? ""})` }],
            details: { todoId: options.todoId, revision: state.revision },
          };
        },
      }));
    },
  };
}

export function isStartRequest(request: { action: string }): request is TodoStartRequest {
  return request.action === "start";

export type TodoStartResult =

  | { ok: false; status: number; reason: string; error: string };

export interface TodoStartCaller {
  /** 项目目录是否还在。 */
  exists(cwd: string): boolean;
  /** 该会话是否正在运行（启动第二个会话会把同一个待办跑两遍）。 */

/** 测试可以替换这三件事；生产用下面的默认实现。 */
export function createStartTodo(caller: TodoStartCaller) {
  return async function startTodo(id: string): Promise<TodoStartResult> {
    const state = readTodos();
    const todo = state.todos.find((entry) => entry.id === id);
    if (!todo) return { ok: false, status: 404, reason: "not-found", error: "No such todo" };
    if (!caller.exists(todo.project)) {
      return { ok: false, status: 400, reason: "cwd-missing", error: `The project folder no longer exists: ${todo.project}` };
    }
    if (todo.sessionId && caller.isRunning(todo.sessionId)) {
      return { ok: true, sessionId: todo.sessionId, state };
    }
    const { sessionId } = await caller.startSession({ cwd: todo.project, message: todoStartPrompt(todo), todoId: todo.id });
    const next = await updateTodos({ action: "attach", id: todo.id, sessionId });
    return { ok: true, sessionId, state: next };
  };
}

export const startTodo = createStartTodo({
  exists: (cwd) => existsSync(cwd),
  isRunning: (sessionId) => isSessionRunning(sessionId),
  startSession: async ({ cwd, message, todoId }) => {
    const tempKey = `__todo__${randomUUID()}`;
    const { session, realSessionId } = await startRpcSession(tempKey, "", cwd, { todoId });
    allowFileRoot(cwd);
    invalidateSessionListCache();
    await session.send({ type: "prompt", message });
    return { sessionId: realSessionId };
  },
});

  const wrapper = getRunningWrapper(sessionId);

"use client";

// 待办面板的数据层：一次一个动作，服务端回整个列表，客户端不做乐观合并
// （列表很小；revision 单调，后到的答案即可覆盖先到的）。

import { useCallback, useEffect, useMemo, useState } from "react";
import type { Todo } from "@/lib/todos-shared";

export type TodoAction =
  | { kind: "add"; title: string; project: string; note?: string }
  | { kind: "update"; id: string; title?: string; note?: string }
  | { kind: "complete"; id: string }
  | { kind: "reopen"; id: string }
  | { kind: "delete"; id: string }
  | { kind: "start"; id: string };

export function todoRequestBody(action: TodoAction): Record<string, unknown> {
  const { kind, ...rest } = action;
  return { action: kind, ...rest };
}

export function todoErrorMessage(status: number, body: unknown): string {
  const message = (body as { error?: unknown } | null)?.error;
  return typeof message === "string" && message.length > 0 ? message : `HTTP ${status}`;
}

export interface TodosApi {
  todos: Todo[];
  revision: number;
  loading: boolean;
  error: string | null;
  refresh(): Promise<void>;
  run(action: TodoAction): Promise<{ sessionId?: string }>;
}

export function useTodos(): TodosApi {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [revision, setRevision] = useState(0);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const refresh = useCallback(async () => {
    try {
      const res = await fetch("/api/todos", { cache: "no-store" });
      if (!res.ok) throw new Error(todoErrorMessage(res.status, await res.json().catch(() => null)));
      const body = await res.json() as { todos: Todo[]; revision: number };
      setTodos(body.todos);
      setRevision(body.revision);
      setError(null);
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : String(caught));
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { void refresh(); }, [refresh]);

  const run = useCallback(async (action: TodoAction) => {
    try {
      const res = await fetch("/api/todos", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(todoRequestBody(action)),
      });
      const body = await res.json().catch(() => null);
      if (!res.ok) {
        setError(todoErrorMessage(res.status, body));
        return {};
      }
      const answered = body as { todos: Todo[]; revision: number; sessionId?: string };
      setTodos(answered.todos);
      setRevision(answered.revision);
      setError(null);
      return answered.sessionId ? { sessionId: answered.sessionId } : {};
    } catch (caught) {
      setError(caught instanceof Error ? caught.message : String(caught));
      return {};
    }
  }, []);

  return useMemo(() => ({ todos, revision, loading, error, refresh, run }), [todos, revision, loading, error, refresh, run]);
}
```

// 展示态来自 lib/todos-shared.ts 的 todoDisplayStatus()，面板自己不持久化任何状态。

import { useCallback, useMemo, useState } from "react";
import { useI18n } from "@/hooks/useI18n";
import type { TodoAction } from "@/hooks/useTodos";
import { todoDisplayStatus, type Todo, type TodoDisplayStatus } from "@/lib/todos-shared";

export interface TodoRowModel { todo: Todo; display: TodoDisplayStatus }
export interface TodoGroupModel { project: string; label: string; rows: TodoRowModel[] }

/** 按项目分组：未完成的在前（进行中、待确认、未开始），已完成的沉底。 */
export function groupTodos(
  todos: readonly Todo[],
  runningSessionIds: ReadonlySet<string>,
  projectNames: Record<string, string>,
): TodoGroupModel[] {
  const order: Record<TodoDisplayStatus, number> = { running: 0, confirm: 1, open: 2, done: 3 };
  const groups = new Map<string, TodoGroupModel>();
  for (const todo of todos) {
    const display = todoDisplayStatus(todo, runningSessionIds);
    const group = groups.get(todo.project)
      ?? { project: todo.project, label: projectNames[todo.project] ?? [...todo.project.split(/[\\/]/)].filter(Boolean).pop() ?? todo.project, rows: [] };
    group.rows.push({ todo, display });
    groups.set(todo.project, group);
  }
  return [...groups.values()].map((group) => ({
    ...group,
    rows: [...group.rows].sort((a, b) => {
      const byState = order[a.display] - order[b.display];
      if (byState !== 0) return byState;
      return (b.todo.doneAt ?? b.todo.updatedAt) - (a.todo.doneAt ?? a.todo.updatedAt);
    }),
  }));
}
